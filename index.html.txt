        function isTaskOverdue(dateStr, status) {
            if (status === 'Hoàn thành' || !dateStr) return false;

            const parts = dateStr.split('/');
            if (parts.length !== 3) return false;

            const day = parseInt(parts[0], 10);
            const month = parseInt(parts[1], 10) - 1; // Trong JS Tháng bắt đầu từ 0
            const year = parseInt(parts[2], 10);

            const dueDate = new Date(year, month, day, 23, 59, 59);
            const now = new Date();

            return dueDate < now;
        }

        function editKPICriteria(index) {
            const item = kpiList[index];
            const newTitle = prompt("Sửa tên tiêu chí:", item.title);
            if (newTitle === null) return;

            const newMaxScoreInput = prompt("Sửa điểm tối đa cho tiêu chí:", item.maxScore);
            const newMaxScore = Number(newMaxScoreInput) > 0 ? Number(newMaxScoreInput) : item.maxScore;

            if (newTitle.trim() !== '') item.title = newTitle.trim();
            item.maxScore = newMaxScore;
            if (item.selfScore > newMaxScore) item.selfScore = newMaxScore;

        function updateKPISelfScore(index, value) {
            let val = Number(value);
            if (isNaN(val)) val = 0;
            if (val < 0) val = 0;
            if (val > kpiList[index].maxScore) val = kpiList[index].maxScore;

            kpiList[index].selfScore = val;
            saveKPIRatingData();
            calcKPITotal();
        }

        function calcKPITotal() {
            let selfTotal = 0;
            let maxTotal = 0;

            kpiList.forEach(item => {
                selfTotal += Number(item.selfScore || 0);
                maxTotal += Number(item.maxScore || 0);
            });

            document.getElementById('kpi-total-self').innerText = selfTotal.toFixed(1);
            document.getElementById('kpi-total-max').innerText = maxTotal;
        }

            const dateStr = String(now.getDate()).padStart(2, '0') + '/' + 
                            String(now.getMonth() + 1).padStart(2, '0') + '/' + 
                            now.getFullYear();
            const timeStr = String(now.getHours()).padStart(2, '0') + ':' + 
                            String(now.getMinutes()).padStart(2, '0') + ':' + 
                            String(now.getSeconds()).padStart(2, '0');
            
            const fullTimestamp = `${timeStr} - ${dateStr}`;

            saveKPIRatingData(fullTimestamp);
            alert(`Đã lưu kết quả đánh giá KPI cho [${kpiTargetUser}] thành công lúc ${fullTimestamp}!`);
            renderKPIEvalHistory();
        }

        // --- RENDER BẢNG TỔNG HỢP CÁC KẾT QUẢ ĐÁNH GIÁ MÀ kpiTargetUser NHẬN ĐƯỢC ---
        function renderKPIEvalHistory() {
            document.getElementById('kpi-summary-username').innerText = kpiTargetUser;
            const historyBody = document.getElementById('kpi-eval-history-body');
            historyBody.innerHTML = '';

            const allUsers = getAllUsernames();
            let totalScoresReceived = 0;
            let totalEvaluations = 0;

            allUsers.forEach(evaluator => {
                const storageKey = `kpis_CHAM_${kpiTargetUser}_${evaluator}`;
                const savedData = localStorage.getItem(storageKey);

                if (savedData) {
                    const parsed = JSON.parse(savedData);
                    let items = [];
                    let timestampDisplay = 'Chưa có thông tin';

                    // Xử lý đọc đúng định dạng dữ liệu (Mới/Cũ)
                    if (Array.isArray(parsed)) {
                        items = parsed;
                    } else {
                        items = parsed.kpiList || [];
                        timestampDisplay = parsed.timestamp || 'Chưa rõ';
                    }

                    let userSum = 0;
                    let maxSum = 0;

                    items.forEach(item => {
                        userSum += Number(item.selfScore || 0);
                        maxSum += Number(item.maxScore || 0);
                    });

                    totalScoresReceived += userSum;
                    totalEvaluations++;

                    const roleTag = (evaluator === ADMIN_USERNAME) 
                        ? `<span style="background-color: #dcfce7; color: #15803d; padding: 2px 8px; border-radius: 4px; font-size: 11px; font-weight: bold;">ADMIN</span>` 
                        : `<span style="background-color: #dbeafe; color: #1d4ed8; padding: 2px 8px; border-radius: 4px; font-size: 11px;">User</span>`;

                    const selfTag = (evaluator === kpiTargetUser) ? ` <small style="color: #64748b;">(Tự chấm)</small>` : '';

                    historyBody.innerHTML += `
                        <tr>
                            <td><b>${evaluator}</b> ${selfTag}</td>
                            <td>${roleTag}</td>
                            <td style="text-align: center; font-weight: bold; color: #2563eb; font-size: 15px;">${userSum.toFixed(1)}</td>
                            <td style="text-align: center; color: #059669; font-weight: bold;">${maxSum}</td>
                            <td style="text-align: center; font-size: 13px; color: #475569;">
                                <i class="fa-regular fa-clock" style="color: #0284c7;"></i> ${timestampDisplay}
                            </td>
                            <td style="text-align: center;">
                                <button class="btn-sm btn-view" onclick="viewDetailedKPI('${evaluator}')"><i class="fa-solid fa-eye"></i> Xem phiếu</button>
                            </td>
                        </tr>
                    `;
                }
            });

            if (totalEvaluations === 0) {
                historyBody.innerHTML = `<tr><td colspan="6" style="text-align: center; color: #64748b; padding: 15px;">Chưa có tài khoản nào thực hiện đánh giá cho [${kpiTargetUser}].</td></tr>`;
                document.getElementById('kpi-average-score').innerText = '0';
            } else {
                const avg = totalScoresReceived / totalEvaluations;
                document.getElementById('kpi-average-score').innerText = avg.toFixed(1);
            }
        }

            const storageKey = `kpis_CHAM_${kpiTargetUser}_${evaluatorUsername}`;

        }

        // Cập nhật biểu đồ
        function updateCharts() {
            if (!statusChartInstance || !priorityChartInstance) return;

            const dataSource = (currentUser === ADMIN_USERNAME) ? getAllUsersTasks() : tasks;

            let todo = dataSource.filter(t => t.status === 'Chưa làm').length;
            let doing = dataSource.filter(t => t.status === 'Đang làm').length;
            let done = dataSource.filter(t => t.status === 'Hoàn thành').length;

            statusChartInstance.data.labels = [`Chưa làm (${todo})`, `Đang làm (${doing})`, `Hoàn thành (${done})`];
            statusChartInstance.data.datasets[0].data = [todo, doing, done];
            statusChartInstance.update();

            let high = dataSource.filter(t => t.priority === 'Cao').length;
            let normal = dataSource.filter(t => t.priority === 'Bình thường').length;
            let low = dataSource.filter(t => t.priority === 'Thấp').length;

            priorityChartInstance.data.datasets[0].data = [high, normal, low];
            priorityChartInstance.update();
        }

            const ctx1 = document.getElementById('statusChart').getContext('2d');
            const ctx2 = document.getElementById('priorityChart').getContext('2d');

            if (statusChartInstance) statusChartInstance.destroy();
            if (priorityChartInstance) priorityChartInstance.destroy();

            const dataSource = (currentUser === ADMIN_USERNAME) ? getAllUsersTasks() : tasks;

            const scopeLabel = document.getElementById('chart-scope-label');
            if (scopeLabel) {
                scopeLabel.innerText = (currentUser === ADMIN_USERNAME) ? '(Thống kê tổng hợp toàn hệ thống)' : '(Thống kê cá nhân)';
            }

            let todo = dataSource.filter(t => t.status === 'Chưa làm').length;
            let doing = dataSource.filter(t => t.status === 'Đang làm').length;
            let done = dataSource.filter(t => t.status === 'Hoàn thành').length;

            statusChartInstance = new Chart(ctx1, {
                type: 'doughnut',
                data: {
                    labels: [`Chưa làm (${todo})`, `Đang làm (${doing})`, `Hoàn thành (${done})`],
                    datasets: [{
                        data: [todo, doing, done],
                        backgroundColor: ['#a5a6f6', '#2563eb', '#06b6d4'],

                        borderColor: '#ffffff',

                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { position: 'top', labels: { usePointStyle: true, boxWidth: 12, padding: 15, font: { size: 13 } } }
                    }
                }
            });

            let high = dataSource.filter(t => t.priority === 'Cao').length;
            let normal = dataSource.filter(t => t.priority === 'Bình thường').length;
            let low = dataSource.filter(t => t.priority === 'Thấp').length;

            priorityChartInstance = new Chart(ctx2, {
                type: 'bar',
                data: {
                    labels: ['Cao', 'Bình thường', 'Thấp'],

                        borderRadius: 4,
                        barThickness: 60

                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        y: { beginAtZero: true, ticks: { stepSize: 1, precision: 0 }, grid: { color: '#f1f5f9' } },
                        x: { grid: { display: false } }
                    }
                }
            });
        }
